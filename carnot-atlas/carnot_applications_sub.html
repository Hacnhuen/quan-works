# -*- coding: utf-8 -*-
"""
案例① 燃煤超超临界电站净效率核算（LHV 口径）
==============================================
报告依据：《卡诺热机深度研究报告》§4.2 与 §4.8
  * 亚临界      净效率约 33-37%（LHV）
  * 超临界      净效率约 38-42%（LHV）
  * 超超临界    净效率约 42-46%（LHV，600℃ 级主蒸汽）
  * 700℃ A-USC 约 50%（设计目标，瓶颈：镍基高温合金）
  * 同界限卡诺上限（600℃ / 25℃）≈ 66%
  * USC 达卡诺比例约 65-70%
口径纪律（对应可复现性清单第 1、2、4 项）：
  * 一律使用 LHV（低位发热量），HHV 比 LHV 高约 6-10%；
  * 一律使用净效率（扣除厂用电），毛效率会虚高 2-4 个百分点；
  * 必须标注机组年份与蒸汽参数，跨文献对比前先统一口径。
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class PlantInput:
    """电厂边界输入参数（示例：1000 MW 级超超临界机组）"""
    coal_lhv_mj_kg: float = 23.0     # 燃煤低位发热量 LHV（MJ/kg）
    coal_flow_t_h: float = 360.0     # 给煤量（t/h）
    gross_power_mw: float = 1060.0   # 毛发电功率（MW）
    aux_power_mw: float = 48.0       # 厂用电（MW，约占总发电 4.5%）
    main_steam_c: float = 600.0      # 主蒸汽温度（℃）
    condenser_c: float = 25.0        # 冷端温度（℃）
    year: int = 2020                 # 投运年份（口径标注用）


def fuel_power_mw(lhv_mj_kg: float, flow_t_h: float) -> float:
    """由给煤量与 LHV 计算燃料输入热功率（MW）。
    单位换算：1 MJ/s = 1 MW；t/h 换算为 kg/s 需除以 3.6。"""
    return lhv_mj_kg * flow_t_h * 1000.0 / 3600.0


def net_power_mw(gross: float, aux: float) -> float:
    """净功率 = 毛功率 − 厂用电。净口径是行业对比的前提。"""
    return gross - aux


def net_eta_lhv(fuel_mw: float, net_mw: float) -> float:
    """净效率（LHV 口径）= 净电功率 / 燃料输入热功率。"""
    return net_mw / fuel_mw


def to_hhv_basis(eta_lhv: float, hhv_ratio: float = 1.07) -> float:
    """HHV 口径换算：η_HHV ≈ η_LHV · (LHV/HHV)。
    同一种煤 HHV 比 LHV 高约 7%，故 HHV 口径效率数值更低。"""
    return eta_lhv / hhv_ratio


def carnot_eta(steam_c: float, condenser_c: float) -> float:
    """同界限卡诺上限 η = 1 − Tc/Th（必须用绝对温度 Kelvin）。"""
    th = steam_c + 273.15
    tc = condenser_c + 273.15
    return 1.0 - tc / th


def ratio_to_carnot(eta: float, eta_carnot: float) -> float:
    """达卡诺比例 = 实际效率 / 同界限卡诺上限。"""
    return eta / eta_carnot


def main() -> None:
    """核算主流程：算净效率 → 换算 HHV → 对照卡诺上限。"""
    p = PlantInput()
    fuel = fuel_power_mw(p.coal_lhv_mj_kg, p.coal_flow_t_h)
    net = net_power_mw(p.gross_power_mw, p.aux_power_mw)
    eta_net_lhv = net_eta_lhv(fuel, net)
    eta_net_hhv = to_hhv_basis(eta_net_lhv)
    eta_carnot = carnot_eta(p.main_steam_c, p.condenser_c)
    reach = ratio_to_carnot(eta_net_lhv, eta_carnot)

    print("=" * 58)
    print("燃煤超超临界电站净效率核算（LHV 口径）")
    print("=" * 58)
    print(f"  给煤量          : {p.coal_flow_t_h:8.1f} t/h")
    print(f"  燃煤 LHV        : {p.coal_lhv_mj_kg:8.1f} MJ/kg")
    print(f"  燃料输入热功率  : {fuel:10.1f} MW")
    print(f"  毛功率 / 厂用电 : {p.gross_power_mw:8.1f} / {p.aux_power_mw:6.1f} MW")
    print(f"  净功率          : {net:10.1f} MW")
    print("-" * 58)
    print(f"  净效率（LHV）   : {eta_net_lhv*100:8.2f} %")
    print(f"  净效率（HHV换算）: {eta_net_hhv*100:7.2f} %  （口径差异演示）")
    print(f"  卡诺上限（600/25℃）: {eta_carnot*100:6.2f} %")
    print(f"  达卡诺比例      : {reach*100:8.1f} %")
    print("=" * 58)
    if 0.42 <= eta_net_lhv <= 0.46:
        print("  判定：落入超超临界 42-46% 报告区间 ✅")
    else:
        print("  判定：偏离报告区间，请复核输入口径 ⚠️")


if __name__ == "__main__":
    # 内置自检：核对单位换算与效率区间（防止口径漂移）
    assert abs(fuel_power_mw(23.0, 360.0) - 2300.0) < 1e-9
    assert abs(net_power_mw(1060.0, 48.0) - 1012.0) < 1e-9
    assert 0.42 <= net_eta_lhv(2300.0, 1012.0) <= 0.46
    assert 0.65 <= carnot_eta(600.0, 25.0) <= 0.67
    main()

# -*- coding: utf-8 -*-
"""
案例② 联合循环电厂（CCGT）效率与卡诺上限对比
==============================================
报告依据：《卡诺热机深度研究报告》§4.3 与 §4.8
  * Bouchain 9HA.01  净效率 62.22%（2016 吉尼斯实测认证）
  * Keadby 2        约 63%（设计值，非实测）
  * 9HA.02          64%（厂商宣称，未获独立实测证实）
  * 同界限卡诺上限   透平初温约 1500℃ / 排气 25℃ → 约 83%
  * 联合循环已达卡诺约 75%——差距在材料（镍基合金 + TBC），
    不在物理定律。引用效率必须分级：实测 / 设计值 / 厂商宣称。

本脚本做三件事：
  1. 打印三座标杆电厂的口径分级表；
  2. 计算 1500℃/25℃ 界限的卡诺上限与 62.22% 的达卡诺比例；
  3. 扫描底循环参数网格，找出能复现 62.22% 的组合。
  口径分级 = 可复现性清单第 4、5 项的直接应用。
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class CcgtRecord:
    """联合循环标杆电厂的效率口径分级记录。"""
    name: str                 # 电厂 / 机组型号
    eta_pct: float            # 净效率（%）
    grade: str                # 口径等级：实测 / 设计值 / 厂商宣称
    year: int                 # 相关年份


# 报告 §4.3 口径分级数据：实测 > 设计值 > 厂商宣称
CCGT_RECORDS = (
    CcgtRecord("Bouchain 9HA.01", 62.22, "实测（吉尼斯 2016 认证）", 2016),
    CcgtRecord("Keadby 2", 63.00, "设计值（未达实测阶段）", 2020),
    CcgtRecord("9HA.02", 64.00, "厂商宣称（无独立实测）", 2018),
)


def carnot_bound(turbine_inlet_c: float, sink_c: float) -> float:
    """同界限卡诺上限：η = 1 − Tc/Th，温度须用 Kelvin。"""
    th = turbine_inlet_c + 273.15
    tc = sink_c + 273.15
    return 1.0 - tc / th


def ccgt_eta(gt_eta: float, hrsg_recovery: float, st_eta: float) -> float:
    """联合循环效率拆解：
    η_cc = η_gt + (1 − η_gt) · η_hrsg · η_st
    燃机排气的火用先经余热锅炉（HRSG）回收，再由汽轮机
    （底循环）做功，这就是「燃气-蒸汽联合」的能量账。"""
    return gt_eta + (1.0 - gt_eta) * hrsg_recovery * st_eta


def find_combos(target: float, tol: float = 0.001):
    """扫描典型参数网格，找出能复现目标效率的组合。"""
    hits = []
    for gt in (i / 1000 for i in range(380, 431)):       # 0.380-0.430
        for hr in (j / 100 for j in range(78, 93)):      # 0.78-0.92
            for st in (k / 1000 for k in range(350, 431)):  # 0.350-0.430
                if abs(ccgt_eta(gt, hr, st) - target) < tol:
                    hits.append((gt, hr, st))
                    if len(hits) >= 5:                   # 只取前 5 组示例
                        return hits
    return hits


def main() -> None:
    """主流程：口径分级表 → 卡诺上限 → 达卡诺比例 → 参数复现。"""
    print("=" * 64)
    print("联合循环标杆电厂效率（口径分级）")
    print("=" * 64)
    for rec in CCGT_RECORDS:
        print(f"  {rec.name:<16s} {rec.eta_pct:6.2f} %   [{rec.grade}]  {rec.year}")

    bound = carnot_bound(1500.0, 25.0)
    ratio = 62.22 / 100.0 / bound
    print("-" * 64)
    print(f"  卡诺上限（1500℃ / 25℃）      : {bound*100:6.2f} %")
    print(f"  Bouchain 达卡诺比例           : {ratio*100:6.1f} %")
    print(f"  结论：联合循环已达同界限卡诺约 75%——瓶颈在材料。")

    print("-" * 64)
    print("底循环参数复现 62.22%（η_gt, η_hrsg, η_st）示例：")
    for gt, hr, st in find_combos(0.6222):
        print(f"    GT {gt:.3f} · HRSG 回收 {hr:.2f} · ST {st:.3f}"
              f"  →  {ccgt_eta(gt, hr, st)*100:.2f} %")
    print("=" * 64)


if __name__ == "__main__":
    # 内置自检：卡诺上限 1500/25 应在 0.83 附近
    assert abs(carnot_bound(1500.0, 25.0) - 0.8318) < 0.005
    assert abs(ccgt_eta(0.40, 0.85, 0.35) - 0.5785) < 1e-9
    assert find_combos(0.6222, 0.01)  # 网格内必能命中
    main()

案例③ 压水堆核电站（PWR）热效率计算
====================================
报告依据：《卡诺热机深度研究报告》§4.2 与 §4.8
  * 压水堆核电站热效率约 33%（区间 30-37%）
  * 蒸汽参数较低：约 285℃ / 7 MPa（饱和蒸汽）
  * 同界限卡诺上限：285℃ / 冷端 30℃ → η ≈ 46%
  * 达卡诺比例约 72%
为什么核电站效率低于燃煤？
  * 堆芯不能像锅炉那样烧到 600℃+（燃料包壳/堆内材料限制），
    蒸汽参数被压在 285℃ 附近 → 卡诺上限本身就低；
  * 效率 = 卡诺上限 × 达卡诺比例：核电站两者都不占优；
  * 约 2/3 的反应堆热功率必须由冷凝器排给环境（冷却水）。
本脚本：
  1. 由蒸汽表锚点估算 7 MPa 饱和温度；
  2. 给定堆热功率核算净电功率与净效率；
  3. 对照卡诺上限，核算排热功率与达卡诺比例。
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class PwrInput:
    """压水堆核岛-常规岛边界参数（示例：百万千瓦级机组）"""
    reactor_thermal_mw: float = 3400.0   # 反应堆热功率（MWth）
    gross_electric_mw: float = 1190.0    # 毛电功率（MWe）
    aux_power_mw: float = 68.0           # 厂用电（MWe）
    steam_pressure_bar: float = 70.0     # 蒸汽压力（7 MPa = 70 bar）
    condenser_c: float = 30.0            # 冷端（循环水）温度
    year: int = 2019                     # 机组投运年份


def saturation_temp_c(p_bar: float) -> float:
    """7 MPa 附近水饱和温度：蒸汽表锚点 + 局部线性化。
    锚点：70 bar ↔ 285.8℃（国际汽水性质表，IAPWS-IF97）；
    斜率：约 1.1 ℃/bar（7 MPa 邻域）。
    说明：高压区 Antoine 相关式精度不足，教材脚本用锚点
    线性化；工程核算请直接查 IAPWS-IF97 或 NIST 蒸汽表。"""
    return 285.8 + 1.1 * (p_bar - 70.0)


def net_electric_mw(gross: float, aux: float) -> float:
    """净电功率 = 毛电功率 − 厂用电。"""
    return gross - aux


def thermal_efficiency(net_mwe: float, thermal_mw: float) -> float:
    """热效率 = 净电功率 / 堆热功率。"""
    return net_mwe / thermal_mw


def carnot_eta(steam_c: float, condenser_c: float) -> float:
    """同界限卡诺上限 η = 1 − Tc/Th（绝对温度）。"""
    th = steam_c + 273.15
    tc = condenser_c + 273.15
    return 1.0 - tc / th


def rejected_heat_mw(thermal_mw: float, eta: float) -> float:
    """冷凝器排热 = Qh × (1 − η)——第一定律的直接推论。"""
    return thermal_mw * (1.0 - eta)


def main() -> None:
    """核算主流程：饱和温度 → 净效率 → 卡诺对比 → 排热。"""
    p = PwrInput()
    tsat = saturation_temp_c(p.steam_pressure_bar)
    net = net_electric_mw(p.gross_electric_mw, p.aux_power_mw)
    eta = thermal_efficiency(net, p.reactor_thermal_mw)
    bound = carnot_eta(tsat, p.condenser_c)
    ratio = eta / bound
    qc = rejected_heat_mw(p.reactor_thermal_mw, eta)

    print("=" * 60)
    print("压水堆核电站热效率核算")
    print("=" * 60)
    print(f"  蒸汽压力          : {p.steam_pressure_bar:6.1f} bar")
    print(f"  饱和温度（估算）  : {tsat:8.1f} ℃")
    print(f"  堆热功率          : {p.reactor_thermal_mw:8.0f} MWth")
    print(f"  毛 / 净电功率     : {p.gross_electric_mw:8.0f} / {net:6.0f} MWe")
    print(f"  热效率            : {eta*100:8.2f} %   （报告约 33%）")
    print("-" * 60)
    print(f"  卡诺上限（{tsat:.0f}℃/30℃）  : {bound*100:8.2f} %")
    print(f"  达卡诺比例        : {ratio*100:8.1f} %   （报告约 72%）")
    print(f"  冷凝器排热        : {qc:8.0f} MWth  ≈ 热功率的 {1-eta:.0%}")
    print("=" * 60)
    if 0.30 <= eta <= 0.37:
        print("  判定：落入核电站 30-37% 报告区间 ✅")


if __name__ == "__main__":
    # 内置自检：7 MPa 饱和温度应约 286℃；效率应约 33%
    assert abs(saturation_temp_c(70.0) - 285.8) < 1e-9
    assert abs(thermal_efficiency(1122.0, 3400.0) - 0.33) < 1e-9

"""
案例④ 家用冰箱 COP：实测 vs 卡诺理想
=====================================
报告依据：《卡诺热机深度研究报告》§4.6 与 §4.8
  * 冰箱冷冻室 −18℃（255 K）/ 环境 25℃（298 K）：
    卡诺制冷 COP = Tc/(Th−Tc) ≈ 5.9
  * 家用冰箱实际 COP 仅约 1.5-3 → 达卡诺约 25-50%
  * 家用空调 EER ≈ 3；27℃/35℃ 工况卡诺 COP ≈ 37 → 约 8%
为什么实际 COP 这么低？
  * 制冷循环是逆卡诺循环 + 真实压缩机的不可逆损失；
  * 蒸发器/冷凝器必须有有限温差才能换热（温差即火用毁）；
  * 温差越大，实际 COP 相对卡诺下滑越明显（§4.6 硬约束）。
本脚本：
  1. 计算任意冷冻温度下的卡诺 COP 并与实测 COP 对比；
  2. 扫描冷冻室温度窗口，观察 COP 随温差的变化；
  3. 核算「每天耗电」并换算全年电费（口径演示）。
"""

from dataclasses import dataclass


@dataclass(frozen=True)
class FridgeInput:
    """冰箱工况与实测参数（示例机型）"""
    freezer_c: float = -18.0        # 冷冻室温度（℃）
    ambient_c: float = 25.0         # 环境温度（℃）
    measured_cop: float = 2.0       # 实测 COP（行业区间 1.5-3）
    daily_cooling_mj: float = 6.0   # 每日需移出热量（MJ，按冷藏负荷估）
    electricity_yuan_kwh: float = 0.55   # 电价（元/kWh）


def kelvin(c: float) -> float:
    """摄氏 → 绝对温度（Kelvin）。"""
    return c + 273.15


def carnot_cop_ref(tc_c: float, th_c: float) -> float:
    """逆卡诺制冷 COP = Tc/(Th−Tc)。"""
    tc = kelvin(tc_c)
    th = kelvin(th_c)
    return tc / (th - tc)


def carnot_cop_hp(tc_c: float, th_c: float) -> float:
    """逆卡诺热泵 COP = Th/(Th−Tc)（供热视角，案例⑥会复用）。"""
    tc = kelvin(tc_c)
    th = kelvin(th_c)
    return th / (th - tc)


def ratio_to_carnot(cop: float, cop_carnot: float) -> float:
    """达卡诺比例 = 实测 COP / 卡诺 COP。"""
    return cop / cop_carnot


def daily_electricity_kwh(cooling_mj: float, cop: float) -> float:
    """日耗电 = 移热量 / COP；1 kWh = 3.6 MJ。"""
    return cooling_mj / 3.6 / cop


def scan_freezer_window(ambient_c: float) -> None:
    """扫描冷冻室温度 −25℃…−10℃，观察卡诺 COP 变化。"""
    print("  冷冻室(℃)   卡诺COP(25℃环境)   相对 −18℃ 变化")
    base = carnot_cop_ref(-18.0, ambient_c)
    for t in range(-25, -9, 5):
        cop = carnot_cop_ref(float(t), ambient_c)
        print(f"   {t:6.0f}℃      {cop:10.2f}         {cop/base:8.1%}")


def main() -> None:
    """主流程：卡诺对比 → 温度扫描 → 电费核算。"""
    f = FridgeInput()
    cc = carnot_cop_ref(f.freezer_c, f.ambient_c)
    ratio = ratio_to_carnot(f.measured_cop, cc)
    kwh = daily_electricity_kwh(f.daily_cooling_mj, f.measured_cop)
    cost = kwh * 365 * f.electricity_yuan_kwh

    print("=" * 60)
    print("家用冰箱 COP：实测 vs 卡诺理想")
    print("=" * 60)
    print(f"  冷冻室 / 环境      : {f.freezer_c:6.1f}℃ / {f.ambient_c:5.1f}℃")
    print(f"  卡诺 COP（制冷）   : {cc:10.2f}   （报告 ≈5.9）")
    print(f"  实测 COP           : {f.measured_cop:10.2f}   （行业 1.5-3）")
    print(f"  达卡诺比例        : {ratio*100:10.1f} %")
    print(f"  日耗电 / 年电费   : {kwh:8.2f} kWh / {cost:8.0f} 元")
    print("-" * 60)
    print("  ★ 顺带：空调 27℃/35℃ 的卡诺 COP 高达")
    print(f"     {carnot_cop_ref(27.0, 35.0):.0f}，而 EER≈3 → 仅约 8%")
    print("-" * 60)
    scan_freezer_window(f.ambient_c)
    print("=" * 60)


if __name__ == "__main__":
    # 内置自检：−18/25℃ 卡诺 COP ≈ 5.93；空调 27/35 ≈ 37.5
    assert abs(carnot_cop_ref(-18.0, 25.0) - 5.93) < 0.02
    assert abs(carnot_cop_ref(27.0, 35.0) - 37.52) < 0.05
    assert abs(ratio_to_carnot(2.0, 5.93) - 0.337) < 0.01
    main()

"""
案例⑤ Otto 循环汽油机效率：压缩比 → 理论 → 现实
=================================================
报告依据：《卡诺热机深度研究报告》§4.4 与 §4.8
  * Otto 循环理论效率：η = 1 − 1/r^(γ−1)
  * 压缩比 r = 14、γ = 1.4 → 理论效率约 65%
  * 量产汽油机最高热效率约 41%（丰田 Dynamic Force，
    厂商口径）；马自达宣称 43%、船用二冲程柴油机约 50%
    均为厂商/行业口径，未经第三方独立认证
  * 达理论（Otto）比例约 63%
为什么理论 65% 只剩 41%？
  * 燃烧强不可逆（火焰传播 = 熵产）；有限温差传热；
  * 排气火用损失（排气仍带高温）；摩擦与泵气损失；
  * 提高压缩比是逼近理论效率的主路径，但受爆震限制。
与卡诺的关系：
  * Otto 效率只依赖 r 与 γ，与温度界限无关；
  * 要问「Otto 距卡诺还有多远」，必须先在同一温度界限下
    计算卡诺效率，再求达卡诺比例（检查清单第 7 项）；
  * 本脚本末尾给出同界限卡诺参照，演示正确的比较姿势。
本脚本：
  1. 计算给定压缩比与 γ 的 Otto 理论效率；
  2. 扫描压缩比 6-16，观察「边际收益递减」；
  3. 拆解 65% → 41% 的损失分配（教学演示）；
  4. 输出同界限卡诺参照，并对比 Otto 与卡诺的差距。
"""


def otto_eta(compression_ratio: float, gamma: float = 1.4) -> float:
    """Otto 循环理论效率 η = 1 − 1/r^(γ−1)。
    注意：与温度界限无关，这是 Otto 循环与卡诺循环的
    本质差别——比较时必须取相同的温度范围（§4.4）。"""
    return 1.0 - 1.0 / (compression_ratio ** (gamma - 1.0))


def scan_compression_ratio(gamma: float = 1.4) -> None:
    """扫描压缩比 6→16：每提高 2 个单位，效率增量递减。"""
    print("  压缩比 r   Otto 理论效率   相邻增量")
    prev = 0.0
    for r in range(6, 17, 2):
        eta = otto_eta(float(r), gamma)
        print(f"    {r:3d}        {eta*100:8.2f} %      {eta-prev:+.3%}")
        prev = eta


def loss_breakdown(theory: float, real: float) -> None:
    """把理论到现实的差距拆成四大损失项（示意分配）。"""
    gap = theory - real
    shares = {

"""

============================================
报告依据：《卡诺热机深度研究报告》§4.6 与 §4.8
  * 空气源热泵 SCOP 约 2.5-4（EN 14825 季节性口径）
  * 0℃/35℃ 供热工况卡诺 COP ≈ 8.8 → 达卡诺约 30-45%
  * 地源热泵 SCOP 约 3.5-4.5（地温稳定，卡诺上限更高）

EN 14825 的核心思想（本脚本按此简化复现）：
  1. 把供暖季按室外温度分成若干「温度 bin」，统计各 bin 小时数；
  2. 每个 bin 用该温度下的 COP 计算耗电（部分负荷修正简化）；
  3. 季节能效 SCOP = 总供热量 / 总耗电（含备用电加热的权重）。
简化假设（工程上需按标准全文执行）：
  * 供热需求近似恒定（真实标准按建筑负荷曲线）；
  * COP 模型 = 卡诺 COP × 品质系数（0.35-0.45，随温差衰减）；
  * 低于双温点（bivalent）时启动辅助电加热（COP = 1）。
"""


# 平均气候区（斯特拉斯堡）供暖季温度 bin 小时数（示意数据）
BIN_HOURS = (
    (-12.0, 40),    # ≤ −12℃ 附近
    (-8.0, 120),
    (-4.0, 300),
    (0.0, 420),
    (4.0, 520),
    (8.0, 380),
    (12.0, 220),
    (15.0, 100),
)


def carnot_cop_hp(outdoor_c: float, supply_c: float) -> float:
    """逆卡诺热泵 COP = Th/(Th−Tc)，供热视角。"""
    tc = outdoor_c + 273.15
    th = supply_c + 273.15
    return th / (th - tc)


def cop_model(outdoor_c: float, supply_c: float, quality: float) -> float:
    """实际 COP = 卡诺 COP × 品质系数（有限温差 + 压缩机损失）。"""
    return carnot_cop_hp(outdoor_c, supply_c) * quality


def bin_hours(outdoor_c: float) -> float:
    """查温度 bin 小时数（锚点间线性插值，简化实现）。"""
    for i in range(len(BIN_HOURS) - 1):
        t0, h0 = BIN_HOURS[i]
        t1, h1 = BIN_HOURS[i + 1]
        if t0 <= outdoor_c <= t1:
            return h0 + (h1 - h0) * (outdoor_c - t0) / (t1 - t0)
    return 0.0


def seasonal_scop(supply_c: float, quality: float,
                  bivalent_c: float = -8.0) -> float:
    """按 EN 14825 思想积分供暖季：SCOP = ΣQ / ΣE。"""
    heat_total = 0.0     # 总供热量（相对单位：kW × h）
    elec_total = 0.0     # 总耗电（kWh）
    # 积分区间：−20℃ 到 +15℃，步长 1℃（数值积分演示）
    t = -20.0
    while t <= 15.0:
        h = bin_hours(t)
        if h <= 0.0:
            t += 1.0
            continue
        heat = 1.0 * h                    # 供热功率取 1 kW
        if t < bivalent_c:                # 低于双温点：电加热补足
            cop = 1.0                     # 电加热 COP = 1
        else:
            cop = cop_model(t, supply_c, quality)
        elec_total += heat / cop
        heat_total += heat
        t += 1.0
    return heat_total / elec_total


def main() -> None:
    """主流程：品质系数扫描 → 季节 SCOP → 卡诺对比。"""
    supply = 35.0   # 供水温度 35℃（低温采暖，热泵友好）
    print("=" * 62)
    print("空气源热泵 SCOP 估算（EN 14825 思路简化）")
    print("=" * 62)
    print(f"  供水温度 35℃；双温点 −8℃；平均气候区 bin 小时")
    print("-" * 62)
    print("  品质系数    SCOP(季节)    报告区间 2.5-4")
    for q in (0.30, 0.35, 0.40, 0.45, 0.50):
        scop = seasonal_scop(supply, q)
        mark = "✅" if 2.5 <= scop <= 4.0 else " "
        print(f"    {q:.2f}       {scop:6.2f}         {mark}")
    print("-" * 62)
    cc = carnot_cop_hp(0.0, 35.0)
    print(f"  0℃/35℃ 卡诺 COP 基准 : {cc:.1f}   （报告 ≈8.8）")
    print(f"  典型 SCOP 3.2 达卡诺   : {3.2/cc*100:.0f} %   （30-45%）")
    print("=" * 62)


if __name__ == "__main__":
    # 内置自检：0/35℃ 卡诺 COP ≈ 8.8；SCOP 落在合理区间
    assert abs(carnot_cop_hp(0.0, 35.0) - 8.80) < 0.05
    scop = seasonal_scop(35.0, 0.40)
    assert 2.0 <= scop <= 5.0
    main()

EPS = 1.0          # 能隙（激发能量）

# -*- coding: utf-8 -*-
"""
案例⑧ 教学诊断：学生「优于卡诺」误解测试评分脚本
==================================================
报告依据：《卡诺热机深度研究报告》§6.2（Smith 等 2015, PRSTPER）
  * 60%（15/25）能正确给出有限热容储热器间卡诺循环的宇宙熵变
  * <30%（19/64）能正确推理「卡诺循环后宇宙熵不变」
  * <20%（11/64）认识到「优于卡诺的引擎」违反热力学定律
  * 统计物理课学生仅 2/5 四问全对；热力学课课后无人四问全对

教学结论（§6.3 误解 8）：
  效率上限与第二定律脱节是最普遍、最顽固的困难——教学设计
  应把卡诺循环讲成第二定律的「几何表达」，而非孤立公式。

本脚本：
  1. 生成一份符合报告通过率分布的模拟答卷（可换真实 CSV）；
  2. 按答题卡评分，输出每题通过率与总体统计；
  3. 按薄弱知识点给出干预建议。
"""

import csv          # 真实答卷数据接入用（本演示用模拟生成器）
import random


# 答题卡（0 = 错，1 = 对）
ANSWER_KEY = {
    "q1": 1,   # Q1 有限热容储热器间宇宙熵变
    "q2": 1,   # Q2 卡诺循环后宇宙熵不变
    "q3": 0,   # Q3 「优于卡诺的引擎」是否合法（合法 = 0 错误！）
    "q4": 1,   # Q4 循环一周后工质熵变
}
# 各题在报告中的通过率（Smith 2015；q4 为教学示意值）
REPORT_PASS = {"q1": 0.60, "q2": 0.30, "q3": 0.20, "q4": 0.60}
N_STUDENTS = 64          # 报告样本量


def gen_responses(n: int, seed: int = 10) -> list:
    """按报告通过率生成模拟答卷（真实场景请读 CSV）。
    注意：每个学生以概率 p 给出「正确答案」，否则给错误答案——
    对 q3 而言正确答案是 0（「优于卡诺合法」= 错），因此
    通过率 ≈ p，与报告 <20% 的实证对齐。"""
    rng = random.Random(seed)
    rows = []
    for sid in range(1, n + 1):
        row = {"student": sid}
        for q, p in REPORT_PASS.items():
            row[q] = ANSWER_KEY[q] if rng.random() < p else 1 - ANSWER_KEY[q]
        rows.append(row)
    return rows


def score(rows: list) -> dict:
    """评分：每题通过率 + 样本量。"""
    stats = {q: [0, 0] for q in ANSWER_KEY}   # [对, 错]
    for row in rows:
        for q, ans in ANSWER_KEY.items():
            if row[q] == ans:
                stats[q][0] += 1
            else:
                stats[q][1] += 1
    result = {}
    for q, (ok, bad) in stats.items():
        result[q] = ok / len(rows)
    result["n"] = len(rows)
    return result


def report(stats: dict) -> None:
    """打印诊断报告并与报告实证数据对照。"""
    print("=" * 62)
    print("教学诊断：'优于卡诺'误解测试评分报告")
    print("=" * 62)
    print(f"  样本量：{stats['n']} 份（报告 n=64）")
    labels = {
        "q1": "宇宙熵变（有限热容储热器）",
        "q2": "卡诺循环后宇宙熵不变",
        "q3": "'优于卡诺'违反第二定律",
        "q4": "循环一周工质熵变为零",
    }
    for q in ANSWER_KEY:
        got = stats[q]
        ref = REPORT_PASS[q]
        flag = "≈报告" if abs(got - ref) < 0.08 else "偏离"
        print(f"  Q 通过率 {got*100:5.1f} %  参考 {ref*100:4.0f} %  "
              f"{flag}  | {labels[q]}")
    p3 = stats["q3"]
    print("-" * 62)
    print(f"  核心困难确认：{p3*100:.0f}% 学生认为'优于卡诺'合法")
    if p3 < 0.20:
        print("  → 与报告 <20% 实证一致：效率上限与第二定律脱节")
    print("  干预建议：")
    print("    1. 用反证法演示：优于卡诺 → 自发把热从冷搬热；")
    print("    2. 把卡诺循环画成 T-S 矩形，讲成第二定律的几何；")
    print("    3. 先讲'宇宙熵增'再回到效率公式，建立因果链。")
    print("=" * 62)


def main() -> None:
    """主流程：生成答卷 → 评分 → 对照报告 → 输出建议。"""
    rows = gen_responses(N_STUDENTS)
    stats = score(rows)
    report(stats)


if __name__ == "__main__":
    # 内置自检：评分器对称性 + 全对答卷通过率 100%
    perfect = [dict({"student": i}, **{q: 1 for q in ANSWER_KEY})
               for i in range(10)]
    assert score(perfect)["q1"] == 1.0
    # 注：q3 的「正确答案」是 0（'优于卡诺'合法 = 错）
    assert ANSWER_KEY["q3"] == 0
    main()

10 项可复现性检查清单

源自《卡诺热机深度研究报告》§8.2 的 12 项矛盾裁决与 §7.4 的 6 项「未证实」边界声明—— 每条清单都是一次真实踩坑的总结。撰写任何「效率 / COP」结论前，逐条对照。

01
LHV / HHV 口径必须声明
口径问题

低位/高位发热量相差约 6-10%。混用会让效率数字整体漂移，跨文献对比前必须统一。 案例①演示同一机组 LHV 44.00% → HHV 41.12%。§4.8 注：净/毛 × LHV/HHV 组合导致 ±2-3pp 差异。

02
净效率 vs 毛效率
口径问题

净效率扣除厂用电（燃煤约 3-5%），毛效率虚高 2-4 个百分点。 「净 + LHV」是行业对比默认口径，任何毛效率数据需换算后再比。§4.2 [20][24]

03
年份标注
纪录属性

效率纪录有年代属性：62.22% 属于 2016 年的 Bouchain 实测；2020 年代的 机组效率更高不矛盾。跨年份比较必须同口径同代际。§4.3 [25]

04
厂商宣称 vs 独立实测
裁决 #1

9HA.02 宣称 64% 无独立实测证实，报告将其降级为「未证实」引用。 厂商数字只进新闻稿，不进基准表——除非标注口径等级。§8.2 裁决 #1 [28]

05
设计值 ≠ 实测值
裁决 #1

Keadby 2 的 63% 是设计值（设计工况、设计燃料），真实运行受环境温度、 负荷率与老化影响。引用时区分「设计值 / 实测值 / 宣称值」三档。§4.3 [27]

06
绝对温度，勿用摄氏代入
公式纪律

η=1−Tc/Th 只对绝对温度成立。用摄氏直接代入会得到荒谬结果—— 这是教材与作业里最高频的错误（§3.6 ③）。案例①-⑦全部显式 +273.15。§3.6 [14]

07
卡诺上限的「同界限」前提
比较纪律

Otto 循环效率与温度界限无关，与卡诺比较必须先取相同温度范围； 联合循环的 83% 卡诺上限基于 1500℃ 初温。界限不同，数字不可直接比。§4.4 [29]

08
循环类别：理论=卡诺 ≠ 实际
误读点

理想 Stirling/Ericsson 理论效率等于同界限卡诺，但回热器压降与温差传热 使其实际显著低于卡诺——「理论等于、实际低于」是评估真实热机的常见误读点。§4.5 [31]

09
卡诺上限是示例计算值
口径声明

基准表中「卡诺上限」列为按公式的示例计算值，不是标定实测； 温度界限取定后数字才唯一。引用时必须写「按 η=1−Tc/Th 计算」。§4.8 注

10
未证实边界声明
反自欺

数据边界与使用声明

本馆全部数值与结论的源头、口径与已知边界——先读这里，再引用。

▸ 必须一起引用的三件事

  • 口径：效率必须带 LHV/HHV × 净/毛 × 示例温度界限；COP 必须带冷端/热端温度与季节口径（SCOP 不混单点 COP）。

▸ 未证实 / 存疑项（§7.4 反自欺清单）

  • 「1878 第二版」年份仅有二手文献支持，未直接核实一手页面；「热是运动」出自 1927 年出版的未刊手稿，非 1824 正文。
